class MusicControls(discord.ui.View):
    def __init__(self, cog):
        super().__init__(timeout=None)
        self.cog = cog
        
    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        vc = interaction.guild.voice_client
        return vc is not None and vc.channel == interaction.user.voice.channel if interaction.user.voice else False
        
    @discord.ui.button(label="⏸", style=discord.ButtonStyle.primary)
    async def pause_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc: discord.VoiceClient = interaction.guild.voice_client
        try:
            if vc.is_playing():
                vc.pause()
                button.label = "▶"
                button.style = discord.ButtonStyle.success
                await interaction.response.edit_message(view=self)
            await interaction.response.send_message(f"Paused by : {interaction.user.mention}", delete_after=5)
        except Exception as e:
            await interaction.response.send_message(f"Interaction Failed : {e}", ephemeral=True, delete_after=3)
            cogs_logger.warning(f"Failed interaction in [Pause Button] : {e}")
            pass

    @discord.ui.button(label="▶", style=discord.ButtonStyle.success, disabled=True)
    async def resume_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc: discord.VoiceClient = interaction.guild.voice_client
        try:
            if vc.is_paused():
                vc.resume()
                self.children[0].label = "⏸"
                self.children[0].style = discord.ButtonStyle.primary
                await interaction.response.edit_message(view=self)
            await interaction.response.send_message(f"Resumed by : {interaction.user.mention}", delete_after=5)
        except Exception as e:
            await interaction.response.send_message(f"Interaction Failed : {e}", ephemeral=True, delete_after=3)
            cogs_logger.warning(f"Failed interaction in [Resume Button] : {e}")
            pass

    @discord.ui.button(label="⏭", style=discord.ButtonStyle.primary)
    async def skip_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc: discord.VoiceClient = interaction.guild.voice_client
        try:
            if vc.is_playing() or vc.is_paused():
                vc.stop()
                await interaction.response.send_message(f"{interaction.user.mention} Skipped the song.", delete_after=5)
            else:
                await interaction.response.send_message("Nothing is playing!", ephemeral=True, delete_after=4)
        except Exception as e:
            await interaction.response.send_message(f"Interaction Failed : {e}", ephemeral=True, delete_after=5)
            cogs_logger.warning(f"Failed interaction in [Skip Button] : {e}")
            pass

    @discord.ui.button(label="⏹", style=discord.ButtonStyle.danger)
    async def stop_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc: discord.VoiceClient = interaction.guild.voice_client
        guild_id = interaction.guild.id
        try:
            self.cog.queues[guild_id].clear()
            if guild_id in self.cog.current:
                del self.cog.current[guild_id]
            vc.stop()
            self.pause_button.disabled = True
            self.resume_button.disabled = True
            self.skip_button.disabled = True
            await interaction.response.edit_message(view=self)
            await interaction.response.send_message(f"{interaction.user.mention} Stopped the song! song queue deleted!", delete_after=5)
        except Exception as e:
            await interaction.response.send_message(f"Interaction Failed : {e}", ephemeral=True, delete_after=5)
            cogs_logger.warning(f"Failed interaction in [Stop Button] : {e}")
            pass
            
        try:
            await presence.change_presence(bot=self.cog, activity_type= discord.ActivityType.watching, activity="I'm Online")
        except Exception as e:
            cogs_logger.warning(f"Failed to change presence in Stop button : {e}")
            pass

    @discord.ui.button(label="Autoplay: 🔴", style=discord.ButtonStyle.secondary)
    async def autoplay_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        gid = interaction.guild.id
        try:
            enabled = not self.cog.autoplay.get(gid, False)
            self.cog.autoplay[gid] = enabled
            button.label = f"Autoplay: {'🟢' if enabled else '🔴'}"
            button.style = discord.ButtonStyle.success if enabled else discord.ButtonStyle.secondary
            await interaction.response.edit_message(view=self)

            if enabled and not self.cog.queues[gid] and interaction.guild.voice_client:
                ctx = await self.cog.bot.get_context(interaction.message)
                await self.cog.start_autoplay_radio(ctx)
        except Exception as e:
            await interaction.response.send_message(f"Interaction Failed : {e}", ephemeral=True, delete_after=5)
            cogs_logger.warning(f"Failed interaction in [Autoplay Button] : {e}")
            pass

    @discord.ui.button(label="☰ Show Queue", style=discord.ButtonStyle.secondary)
    async def show_queue_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        q = self.cog.queues.get(interaction.guild.id, [])
        try:
            if not q:
                return await interaction.response.send_message("Queue Empty!", ephemeral=True, delete_after=5)
            lines = [f"{i+1}. {s['title']}" for i, s in enumerate(q[:15])]
            if len(q) > 15:
                lines.append(f".... and {len(q)-15} more.")
            await interaction.response.send_message("**Current Queue:**\n" + "\n".join(lines))
        except Exception as e:
            await interaction.response.send_message(f"Interaction Failed : {e}", ephemeral=True, delete_after=5)
            cogs_logger.warning(f"Failed interaction in [Autoplay Button] : {e}")
            pass

class Music(commands.Cog):
    def __init__(self, bot):
        self.bot = bot
        self.queues = {}
        self.current = {}
        self.autoplay = {}
        self.recent = {}
        self.disconnect_tasks = {}
        
    def get_queue(self, guild_id: int):
        if guild_id not in self.queues:
            self.queues[guild_id] = []
        return self.queues[guild_id]
        
    async def extract_stream_url(self, url: str) -> str:
        def _extract():
            with yt_dlp.YoutubeDL(YDL_OPTIONS) as ydl:
                info = ydl.extract_info(url=url, download=False)
                return info['url']
        return await asyncio.get_event_loop().run_in_executor(executor=executor, func=_extract)
    
    async def play_next(self, ctx: commands.Context):
        vc: discord.VoiceClient = ctx.voice_client
        if not vc:
            return
        
        guild_id = ctx.guild.id
        queue = self.get_queue(guild_id=guild_id)

        if not queue:
            try:
                await presence.change_presence(bot=self.bot, activity_type=discord.ActivityType.watching, activity="Finding something new...")
                if guild_id in self.current:
                    del self.current[guild_id]
                return
            except Exception as e:
                cogs_logger.warning(f"Something failed in play_next >> 269 - 276 << : {e}")
                return
            
        song = queue.pop(0)
        self.current[guild_id] = song
        
        try:
            stream_url = await self.extract_stream_url(song['url'])
        except Exception as e:
            cogs_logger.error(f"Stram extraction failed >> 283 << : {e}")
            await ctx.send(f"Failed to load {song['title']}. Skipping....")
            return await self.play_next(ctx=ctx)

        source = await discord.FFmpegOpusAudio.from_probe(stream_url, **FFMPEG_OPTIONS)

        view = MusicControls(self)
        view.children[0].disabled = False
        view.children[1].disabled = True
        view.autoplay_button.label = f"Autoplay: {'🟢' if self.autoplay.get(guild_id, False) else '🔴'}"

        embed = discord.Embed(title="Now Playing", color=0x00ff00)
        embed.add_field(name=song['title'], value=f"Duration: {song['duration']}\nViews: {song.get('view_count', 'N/A')}", inline=False)
        embed.set_thumbnail(url=song['thumbnail'])
        embed.set_author(name=song['channel_name'], url=song.get('channel_url'))

        await ctx.send(embed=embed, view=view)

        try:
            await presence.change_presence(bot=self.bot, activity_type=discord.ActivityType.listening, activity=song['title'])
        except Exception as e:
            cogs_logger.warning(f"Failed to change presence in play_next >> 304 << : {e}")
            pass
            
        def after(error):
            if error:
                cogs_logger.error(f"Player error >> 309 << : {e}")
            asyncio.run_coroutine_threadsafe(self.play_next(ctx), self.bot.loop)

        vc.play(source=source, after=after)

        vid = song['video_id']
        recent = self.recent.get(guild_id, [])
        recent = [v for v in recent if v != vid][-9:] + [vid]
        self.recent[guild_id] = recent
        
        try:
            if self.autoplay.get(guild_id, False) and len(queue) < 3:
                await self.add_autoplay_suggestions(ctx=ctx, count=3)
        except Exception as e:
            await ctx.send(f"Failed to add song suggestions : {e}", ephemeral=True, delete_after=5)
            cogs_logger.warning(f"Failed to add autoplay suggestions >> 323 << : {e}")
            pass
        
    async def add_autoplay_suggestions(self, ctx: commands.Context, count: int = 3):
        guild_id = ctx.guild.id
        recent_ids = set(self.recent.get(guild_id, []))
        queued_ids = {s['video_id'] for s in self.get_queue(guild_id)}
        avoid = recent_ids.union(queued_ids)

        last_id = self.recent.get(guild_id, [])[-1] if self.recent.get(guild_id) else None
        if not last_id:
            return
        
        def _get_related():
            with yt_dlp.YoutubeDL({**YDL_OPTIONS, 'extract_flat': True}) as ydl:
                info = ydl.extract_info(f"ytsearch1:https://www.youtube.com/watch?v={last_id}&list=RD{last_id}", download=False)
                return info.get('entries', [])[1:count+10]

        try:
            entries = asyncio.get_event_loop().run_in_executor(executor=executor, func=_get_related)
        except:
            return
        
        added = 0
        for entry in entries:
            if added >= count:
                break
            
            vid = entry['id']
            if vid in avoid or entry.get('title') in ('[Private video]', '[Deleted video]'):
                continue
            info = await self.get_song_info(f"https://www.youtube.com/watch?v={vid}")
            if info:
                self.get_queue(guild_id=guild_id).append(info)
                added += 1
        if added:
            await ctx.send(f"Autoplay added {added} song(s)", delete_after=8)
            
    async def get_song_info(self, query: str) -> dict | None:
        try:
            if query.startswith("http"):
                info = Video.getInfo(query)
            else:
                search = VideosSearch(query=query, limit=1)
                result = search.result()['result']
                if not result:
                    return None
                info = result[0]
                info = Video.getInfo(info['link'])

            duration = info.get('duration', {}).get('secondsText') or info.get('duration') or "LIVE"
            return {
                'url' : info['link'],
                'video_id' : info['id'],
                'title' : info['title'],
                'thumbnail' : info['thumbnails'][0]['url'],
                'channel_name' : info['channel']['title'],
                'channel_url' : info['channel']['link'],
                'duration' : duration if isinstance(duration, str) else f"{duration//60:02d}:{duration%60:02d}",
                'view_count' : info.get('viewCount', {}).get('text', 'N/A'),
            }
        except:
            return None

    async def start_autoplay_radio(self, ctx: commands.Context):
        fallback = "lofi hip hop radio"
        info = await self.get_song_info(fallback)
        if info:
            self.get_queue(ctx.guild.id).append(info)
            await self.play_next(ctx=ctx)

    @commands.Cog.listener()
    async def on_voice_state_update(self, member: discord.Member, before: discord.VoiceState, after: discord.VoiceState):
        guild = member.guild
        voice_client = guild.voice_client

        if member == self.bot.user:
            # Bot was disconnected (e.g., by Discord idle timeout)
            if after.channel is None:
                try:
                    guild_id = guild.id
                    self.get_queue(guild_id).clear() 
                    if guild_id in self.current:
                        del self.current[guild_id]
                    voice_client.stop()
                    
                    if guild.id in self.disconnect_tasks:
                        del self.disconnect_tasks[guild.id]
                        
                    cogs_logger.info(f"Bot disconnected from voice channel in guild {guild.name} ({guild.id})")
                except Exception as e:
                    cogs_logger.error(f"Error handling bot disconnection: {e}")
                    pass
            return

        if voice_client is None:
            return

        channel = voice_client.channel
        if len(channel.members) == 1:  # Only bot left
            if guild.id in self.disconnect_tasks:
                return  # Timer already running
            task = self.bot.loop.create_task(self.disconnect_timer(guild))
            self.disconnect_tasks[guild.id] = task
        else:
            # Users present: cancel any timer
            if guild.id in self.disconnect_tasks:
                self.disconnect_tasks[guild.id].cancel()
                del self.disconnect_tasks[guild.id]
                
    async def disconnect_timer(self, guild: discord.Guild):
        """Timer to disconnect after inactivity when alone"""
        await asyncio.sleep(300)  # 5 minutes
        voice_client = guild.voice_client
        try:
            if voice_client and len(voice_client.channel.members) == 1:
                await voice_client.disconnect(force=True)
                guild.voice_client = None 
        except Exception as e:
            cogs_logger.error(f"Error during disconnect timer for guild {guild.name}: {e}")
            pass
            
            try:
                if guild.id in self.queues:
                    del self.queues[guild.id]
                if guild.id in self.current_songs:
                    del self.current_songs[guild.id]
                if guild.id in self.autoplay:
                    del self.autoplay[guild.id]
                if guild.id in self.recent_played:
                    del self.recent_played[guild.id]
            except Exception as e:
                cogs_logger.error(f"Error cleaning up after disconnect in {guild.name}: {e}")
                pass

        try:
            if guild.id in self.disconnect_tasks:
                del self.disconnect_tasks[guild.id]
            cogs_logger.info(f"Disconnected from voice channel in guild {guild.name} ({guild.id}) due to inactivity.")
        except Exception as e:
            cogs_logger.error(f"Error removing disconnect task for {guild.name}: {e}")
            pass

    @commands.hybrid_command()
    async def arise(self, ctx : commands.Context):            
        """Join the voice channel"""
        if not ctx.author.voice:
            await ctx.send("You are not in a voice channel!")
            return

        channel = ctx.author.voice.channel
        
        try:
            await ctx.defer()
        except Exception:
            pass

        try:
            if ctx.voice_client is not None:
                # NEW: Handle zombie state
                if not ctx.voice_client.is_connected():
                    await ctx.voice_client.disconnect(force=True)
                    ctx.guild.voice_client = None  # Clear zombie

                if ctx.voice_client is not None:  # Re-check after cleanup
                    if ctx.voice_client.channel == channel:
                        await ctx.send("Already in your voice channel!")
                        return
                    else:
                        await ctx.voice_client.move_to(channel)
                        await ctx.send(f"Moved to {channel}")
                        return
        except Exception as e:
            await ctx.send(f"Error handling existing voice client: {e}", ephemeral=True)
            return

        await channel.connect()
        await ctx.send(f"I've been summoned to {channel.name}")
            
    @commands.hybrid_command()
    async def release(self, ctx : commands.Context):
        """ Leave a voice Channel """
        guild_id = ctx.guild.id
        queue = self.get_queue(guild_id)
        if ctx.voice_client:
            queue.clear()
            self.channel = None
            
            try:
                await presence.change_presence(bot= self.bot, activity_type=discord.ActivityType.watching, activity="Back to the begining <<")
            except Exception as e:
                print(f"Error removing bot status >> 521 << : {e}")
                pass
            
            try:
                if guild_id in self.current_songs:
                    del self.current_songs[guild_id]

                await ctx.voice_client.disconnect(force=True)
                try:
                    ctx.guild.voice_client = None
                except Exception:
                    pass
                
                if ctx.guild.id in self.disconnect_tasks:
                    del self.disconnect_tasks[ctx.guild.id]
            except Exception as e:
                await ctx.send(f"Error disconnecting: {e}", ephemeral=True)
                pass
                
            await ctx.send('kbay')
        else:
            await ctx.send("I'm not in a voice channel")
    
    @commands.hybrid_command(name="play")
    @app_commands.describe(
        query="Song title to search on YouTube"
    )
    async def play(self, ctx: commands.Context, query: str):
        """Play a song by query or add to queue"""
        try:
            await ctx.defer()
        except Exception:
            pass
        
        if not ctx.author.voice:
            return await ctx.send("You're not in a voice channel! Please join a voice channel first", ephemeral=True, delete_after=8)
        
        vc: discord.VoiceClient = ctx.guild.voice_client
            
        if not vc:
            vc = await ctx.author.voice.channel.connect()
                
        song = await self.get_song_info(query=query)
        if not song:
            return await ctx.send("No results found.")
        
        queue = self.get_queue(ctx.guild.id)
        if len(queue) >= 20:
            return await ctx.send("Queue is full (max 20)")
        
        queue.append(song)
        if vc.is_playing() or vc.is_paused():
            await ctx.send(f"Added to queue : **{song['title']}**")
        else:
            await self.play_next(ctx)

    @commands.hybrid_command()
    async def lyrics(self, ctx: commands.Context, *, song_query: str = None):
        """Fetch and display lyrics for a song.
        
        Usage:
        - /lyrics [song name and artist] - Fetch lyrics for a specific song
        - /lyrics - Fetch lyrics for the currently playing song
        """
        if not is_genius_available():
            await ctx.send(
                "❌ Lyrics feature is not available. The bot owner needs to set the `GENIUS_API_TOKEN` environment variable.\n"
                "Get a free token from: https://genius.com/api-clients"
            )
            return
        
        guild_id = ctx.guild.id
        
        if not song_query:
            q = self.get_queue(guild_id)
            if not q and guild_id not in self.current_songs:
                await ctx.send("Please provide a song name, or play a song first.")
                return
            if guild_id in self.current_songs:
                song_query = self.current_songs[guild_id].get('title', '')
            else:
                await ctx.send("Please provide a song name, or play a song first.")
                return
        
        try:
            await ctx.defer()
        except Exception:
            pass
        
        try:
            await ctx.send(f"🎵 Searching for lyrics for **{song_query}**...")
            
            result = await fetch_song_lyrics(song_query)
            
            if not result.get('success'):
                await ctx.send(f"❌ {result.get('error', 'Could not fetch lyrics')}")
                return
            
            embed = discord.Embed(
                title=result['title'],
                description=result['artist'],
                url=result['url'],
                color=discord.Color.gold()
            )
            
            lyrics_text = result['lyrics']
            
            if len(lyrics_text) <= character_chunk.MAX_FIELD_LENGTH:
                embed.add_field(name="Lyrics", value=lyrics_text, inline=False)
            else:                    
                chunks = character_chunk.get_chunk(lyrics_text)
                
                for i, chunk in enumerate(chunks):
                    field_name = "Lyrics" if i == 0 else f"Lyrics (cont.)"
                    embed.add_field(name=field_name, value=chunk, inline=False)
            
            embed.set_footer(text="Powered by Genius")
            
            try:
                await ctx.send(embed=embed)
            except Exception as e:
                message_content = f"**{result['title']}** by {result['artist']}\n\n{lyrics_text}\n\n[Full lyrics on Genius]({result['url']})"
                if len(message_content) <= 2000:
                    await ctx.send(message_content)
                else:
                    chunks = [message_content[i:i+1900] for i in range(0, len(message_content), 1900)]
                    for chunk in chunks:
                        await ctx.send(chunk)
        
        except Exception as e:
            await ctx.send(f"❌ Error fetching lyrics: {str(e)}")
        
    @commands.hybrid_command()
    async def show_queue(self, ctx: commands.Context):
        """Show current song queue"""
        guild_id = ctx.guild.id
        queue = self.get_queue(guild_id)
        
        if not queue:
            await ctx.send("The queue is empty.")
            return
        
        queue_list = "\n".join(f"{i+1}. {song['title']}" for i, song in enumerate(queue))
        await ctx.send(f"Current queue:\n{queue_list}")
        
    @commands.hybrid_command()
    async def clear_queue(self, ctx: commands.Context):
        """Clear current queue"""
        guild_id = ctx.guild.id
        self.get_queue(guild_id).clear()
        await ctx.send("Queue cleared.")
        
    @commands.hybrid_command()
    async def skip(self, ctx: commands.Context):
        """Skip current song"""
        if not ctx.voice_client:
            await ctx.send("Not in a voice channel!", ephemeral=True, delete_after=5)
            return
        
        if not ctx.voice_client.is_playing() and not ctx.voice_client.is_paused():
            await ctx.send("Nothing is playing to skip!", ephemeral=True, delete_after=5)
            return
        
        ctx.voice_client.stop()
        await ctx.send(f"{ctx.author.mention} skipped current song!", delete_after=5)
        
    @commands.hybrid_command()
    async def pause(self, ctx : commands.Context):
        """Pause current song"""
        if ctx.voice_client and ctx.voice_client.is_playing():
            ctx.voice_client.pause()
            await ctx.send(f'Paused by {ctx.author.mention}', delete_after=5)
        else:
            await ctx.send('Nothing is currently playing')
    
    @commands.hybrid_command()
    async def resume(self, ctx : commands.Context):
        """Resume current paused song"""
        if ctx.voice_client and ctx.voice_client.is_paused():
            ctx.voice_client.resume()
            await ctx.send(f'Resumed by {ctx.author.mention}', delete_after=5)
        else:
            await ctx.send('Not Paused', ephemeral=True, delete_after=5)
            
    @commands.hybrid_command()
    async def stop(self, ctx : commands.Context):
        """Stop playing song"""
        if ctx.voice_client:
            self.get_queue(ctx.guild.id).clear()
            ctx.voice_client.stop()
            await ctx.send(f"{ctx.author.metion} Stopped the song. Queue cleared!", delete_after=8)
            
            # remove current activity
            try:
                await self.bot.change_presence(activity=discord.Game(name="Hide and seek", platform="Closet"))
            except Exception as e:
                print(f"Error removing bot status: {e}")
                pass
            
            await ctx.send("Stopped")
        else:
            await ctx.send("Nothing to stop")
